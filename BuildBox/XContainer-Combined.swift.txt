// MARK: - Model
// A user script. When isPinned == true it also appears as its own tab/panel.
struct Module: Identifiable, Codable, Hashable {
    var id: UUID = UUID()
    var name: String
    var code: String
    var createdAt: Date = Date()
    var isPinned: Bool = false
    // Which container (folder) this module lives in. Defaults to General
    // so modules saved by older versions still load.
    var folder: String = "General"

    enum CodingKeys: String, CodingKey {
        case id, name, code, createdAt, isPinned, folder
    }

    init(id: UUID = UUID(), name: String, code: String, createdAt: Date = Date(), isPinned: Bool = false, folder: String = "General") {
        self.id = id
        self.name = name
        self.code = code
        self.createdAt = createdAt
        self.isPinned = isPinned
        self.folder = folder
    }

    init(from decoder: Decoder) throws {
        let c = try decoder.container(keyedBy: CodingKeys.self)
        id = try c.decodeIfPresent(UUID.self, forKey: .id) ?? UUID()
        name = try c.decode(String.self, forKey: .name)
        code = try c.decode(String.self, forKey: .code)
        createdAt = try c.decodeIfPresent(Date.self, forKey: .createdAt) ?? Date()
        isPinned = try c.decodeIfPresent(Bool.self, forKey: .isPinned) ?? false
        let rawFolder = try c.decodeIfPresent(String.self, forKey: .folder) ?? "General"
        let trimmed = rawFolder.trimmingCharacters(in: .whitespacesAndNewlines)
        folder = trimmed.isEmpty ? "General" : String(trimmed.prefix(40))
    }
}

// MARK: - Template: copy -> add as module -> use
struct Template: Identifiable, Hashable {
    var id: String { name }
    var name: String
    var category: String
    var blurb: String
    var pinByDefault: Bool = false
    var code: String
}

// MARK: - Run history entry
struct RunEntry: Identifiable, Codable, Hashable {
    var id: UUID = UUID()
    var moduleName: String
    var at: Date = Date()
    var summary: String
    var hadError: Bool = false
}

// MARK: - Store (persisted to UserDefaults so modules survive restarts)
final class ModuleStore: ObservableObject {
    @Published var modules: [Module] = []
    // Set by Modules list when user taps Edit -> Editor picks it up.
    @Published var editingModuleID: UUID? = nil
    @Published var history: [RunEntry] = []

    private let storageKey = "xcontainer.modules.v1"
    private let seededKey = "xcontainer.seeded.v1"
    private let historyKey = "xcontainer.history.v1"

    init() {
        let hadSavedData = load()
        loadHistory()
        // Only seed examples on very first launch. An empty array is a valid
        // saved state (user hit Reset) and must stay empty across restarts.
        if !hadSavedData {
            modules = Self.examples
            save()
            UserDefaults.standard.set(true, forKey: seededKey)
        }
    }

    func recordRun(moduleName: String, summary: String, hadError: Bool) {
        // Auto-runs fire on every tab visit — skip exact repeats so real
        // runs are not pushed out of the 100-entry cap by duplicates.
        if let last = history.first, last.moduleName == moduleName, last.summary == summary, last.hadError == hadError { return }
        let e = RunEntry(moduleName: moduleName, summary: summary, hadError: hadError)
        history.insert(e, at: 0)
        if history.count > 100 { history = Array(history.prefix(100)) }
        saveHistory()
    }

    func clearHistory() {
        history = []
        saveHistory()
    }

    func useTemplate(_ t: Template) -> Module {
        // Templates land in the container matching their category (Games, Tools, …).
        add(name: t.name, code: t.code, pinned: t.pinByDefault, folder: t.category)
    }

    private func loadHistory() {
        guard let data = UserDefaults.standard.data(forKey: historyKey),
              let decoded = try? JSONDecoder().decode([RunEntry].self, from: data) else { return }
        history = decoded
    }

    private func saveHistory() {
        if let data = try? JSONEncoder().encode(history) {
            UserDefaults.standard.set(data, forKey: historyKey)
        }
    }

    func module(withID id: UUID?) -> Module? {
        guard let id else { return nil }
        return modules.first(where: { $0.id == id })
    }

    // Never create two modules with the same visible name.
    func uniqueName(_ base: String) -> String {
        if !modules.contains(where: { $0.name == base }) { return base }
        var i = 2
        while modules.contains(where: { $0.name == base + " (\(i))" }) { i += 1 }
        return base + " (\(i))"
    }

    func add(name: String, code: String, pinned: Bool, folder: String = "General") -> Module {
        let clean = name.trimmingCharacters(in: .whitespacesAndNewlines)
        let safeName = uniqueName(clean.isEmpty ? "Untitled module" : String(clean.prefix(80)))
        let safeCode = String(code.prefix(200_000))
        let f = folder.trimmingCharacters(in: .whitespacesAndNewlines)
        if modules.count >= 500 { modules.removeFirst() }
        let m = Module(name: safeName, code: safeCode, isPinned: pinned, folder: f.isEmpty ? "General" : String(f.prefix(40)))
        modules.append(m)
        save()
        return m
    }

    func update(id: UUID, name: String, code: String, folder: String? = nil) {
        guard let i = modules.firstIndex(where: { $0.id == id }) else { return }
        modules[i].name = name
        modules[i].code = code
        if let f = folder {
            let t = f.trimmingCharacters(in: .whitespacesAndNewlines)
            modules[i].folder = t.isEmpty ? "General" : String(t.prefix(40))
        }
        save()
    }

    func delete(_ module: Module) {
        modules.removeAll(where: { $0.id == module.id })
        if editingModuleID == module.id { editingModuleID = nil }
        save()
    }

    func duplicate(_ module: Module) -> Module {
        if modules.count >= 500 { modules.removeFirst() }
        let copy = Module(name: uniqueName(module.name + " copy"), code: module.code, isPinned: false, folder: module.folder)
        modules.append(copy)
        save()
        return copy
    }

    func importModule(name: String, code: String) -> Module {
        let clean = name.trimmingCharacters(in: .whitespacesAndNewlines)
        if modules.count >= 500 { modules.removeFirst() }
        let m = Module(name: uniqueName(clean.isEmpty ? "Imported module" : String(clean.prefix(80))), code: String(code.prefix(200_000)), isPinned: false)
        modules.append(m)
        save()
        return m
    }

    func togglePin(_ module: Module) {
        guard let i = modules.firstIndex(where: { $0.id == module.id }) else { return }
        modules[i].isPinned.toggle()
        save()
    }

    func setPinned(_ module: Module, pinned: Bool) {
        guard let i = modules.firstIndex(where: { $0.id == module.id }) else { return }
        modules[i].isPinned = pinned
        save()
    }

    func validate(code: String) -> String? {
        let t = code.trimmingCharacters(in: .whitespacesAndNewlines)
        if t.isEmpty { return "Script is empty." }
        var o = 0
        var c = 0
        var inStr: Character? = nil
        var prev: Character? = nil
        for ch in t {
            if let q = inStr {
                if ch == q && prev != "\\" { inStr = nil }
            } else if ch == "\"" || ch == "'" || ch == "`" {
                inStr = ch
            } else if ch == "{" { o += 1 } else if ch == "}" { c += 1 }
            prev = ch
        }
        if inStr != nil { return "Unclosed string literal." }
        if o != c { return "Mismatched braces: \(o) { vs \(c) }." }
        return nil
    }

    func stats() -> (total: Int, pinned: Int, runs: Int, errors: Int) {
        let errs = history.filter { $0.hadError }.count
        return (modules.count, modules.filter { $0.isPinned }.count, history.count, errs)
    }

    func exportJSON() -> URL? {
        let arr = modules.map { ["name": $0.name, "code": $0.code, "pinned": $0.isPinned, "folder": $0.folder, "createdAt": $0.createdAt.timeIntervalSince1970] as [String: Any] }
        guard let data = try? JSONSerialization.data(withJSONObject: arr, options: [.prettyPrinted]) else { return nil }
        let url = FileManager.default.temporaryDirectory.appendingPathComponent("xcontainer-modules.json")
        try? data.write(to: url)
        return url
    }

    func importJSON(_ data: Data) -> Int {
        guard let arr = try? JSONSerialization.jsonObject(with: data) as? [[String: Any]] else { return 0 }
        var n = 0
        for d in arr.prefix(500) {
            guard let code = d["code"] as? String else { continue }
            let name = (d["name"] as? String ?? "Imported").trimmingCharacters(in: .whitespacesAndNewlines)
            let pinned = (d["pinned"] as? Bool) ?? false
            let folder = (d["folder"] as? String ?? "General").trimmingCharacters(in: .whitespacesAndNewlines)
            if modules.count >= 500 { modules.removeFirst() }
            modules.append(Module(name: uniqueName(name.isEmpty ? "Imported" : name), code: String(code.prefix(200_000)), isPinned: pinned, folder: folder.isEmpty ? "General" : folder))
            n += 1
        }
        if n > 0 { save() }
        return n
    }

    func reset() {
        modules = []
        editingModuleID = nil
        save()
    }

    // Adds back any missing examples WITHOUT touching user modules.
    // (It used to overwrite everything — that destroyed work.)
    func restoreExamples() {
        var names = Set(modules.map { $0.name })
        for ex in Self.examples where !names.contains(ex.name) {
            if modules.count >= 500 { modules.removeFirst() }
            modules.append(ex)
            names.insert(ex.name)
        }
        editingModuleID = nil
        save()
    }

    // Container (folder) names currently in use, sorted. Always includes General.
    func folders() -> [String] {
        var set = Set(modules.map { $0.folder })
        set.insert("General")
        return set.sorted()
    }

    // MARK: Persistence
    @discardableResult
    private func load() -> Bool {
        guard let data = UserDefaults.standard.data(forKey: storageKey) else { return false }
        if let decoded = try? JSONDecoder().decode([Module].self, from: data) {
            modules = decoded
            return true
        }
        return false
    }

    private func save() {
        if let data = try? JSONEncoder().encode(modules) {
            UserDefaults.standard.set(data, forKey: storageKey)
        }
    }

// Result of running one script.
struct JSRunResult: Equatable {
    var logs: [String] = []
    var result: String = ""
    var error: String? = nil

    var hasOutput: Bool { !logs.isEmpty || !result.isEmpty || error != nil }
}

// Real JS engine on iOS. Supports console.log(...) / log(...) / print(...)
// and returns the completion value of the script.
private final class LogBox {
    var lines: [String] = []
}

private func formatJSException(_ exc: JSValue) -> String {
    let msg = exc.toString() ?? "Unknown error"
    var line = ""
    var col = ""
    if let l = exc.objectForKeyedSubscript("line")?.toString(), l != "undefined" { line = l }
    if let c = exc.objectForKeyedSubscript("column")?.toString(), c != "undefined" { col = c }
    if !line.isEmpty { return msg + " (line " + line + (col.isEmpty ? "" : ", col " + col) + ")" }
    if let stack = exc.objectForKeyedSubscript("stack")?.toString(), !stack.isEmpty, stack != "undefined" {
        let first = stack.components(separatedBy: "\n").prefix(2).joined(separator: " | ")
        if first.count > 8 { return msg + " | " + String(first.prefix(220)) }
    }
    return msg
}

final class JSEngine {
    static func selfTest() -> String {
        let a = run("1 + 2;").result
        let b = run("console.log('hi'); 'ok';").result
        let c = run("localStorage.setItem('t','1'); localStorage.getItem('t');").result
        return "math=" + a + " log=" + b + " storage=" + c
    }
    static func run(_ code: String) -> JSRunResult {
        var res = JSRunResult()
        // Empty editor should not produce a confusing "undefined" result.
        if code.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty {
            res.result = "(no output)"
            return res
        }
        let box = LogBox()
        var handlerError: String? = nil

        guard let ctx = JSContext() else {
            res.error = "Could not create JS engine"
            return res
        }

        // console.log("a", 1) -> one line "a 1"
        let appendLog: @convention(block) (String) -> Void = { msg in
            box.lines.append(msg)
        }
        ctx.setObject(appendLog, forKeyedSubscript: "iosLog" as NSString)

        // localStorage polyfill backed by UserDefaults (shared across runs).
        // Kept tiny on purpose: string keys/values only.
        let storageKey = "xcontainer.jsStorage.v1"
        let iosGet: @convention(block) (String) -> String = { k in
            let d = UserDefaults.standard.dictionary(forKey: storageKey) as? [String: String] ?? [:]
            return d[k] ?? ""
        }
        let iosHas: @convention(block) (String) -> Bool = { k in
            let d = UserDefaults.standard.dictionary(forKey: storageKey) as? [String: String] ?? [:]
            return d[k] != nil
        }
        let iosSet: @convention(block) (String, String) -> Void = { k, v in
            var d = UserDefaults.standard.dictionary(forKey: storageKey) as? [String: String] ?? [:]
            d[k] = v
            UserDefaults.standard.set(d, forKey: storageKey)
        }
        let iosRemove: @convention(block) (String) -> Void = { k in
            var d = UserDefaults.standard.dictionary(forKey: storageKey) as? [String: String] ?? [:]
            d.removeValue(forKey: k)
            UserDefaults.standard.set(d, forKey: storageKey)
        }
        let iosSleep: @convention(block) (Double) -> Void = { ms in
            if ms > 0 { Thread.sleep(forTimeInterval: min(ms / 1000.0, 2.0)) }
        }
        let iosFetch: @convention(block) (String) -> String = { urlString in
            guard let url = URL(string: urlString) else { return "FETCH_ERROR: bad URL" }
            var out = "FETCH_ERROR: timeout"
            let sem = DispatchSemaphore(value: 0)
            let task = URLSession.shared.dataTask(with: url) { data, _, err in
                defer { sem.signal() }
                if let e = err { out = "FETCH_ERROR: " + e.localizedDescription; return }
                guard let d = data else { out = "FETCH_ERROR: empty"; return }
                out = String(data: d, encoding: .utf8) ?? "FETCH_ERROR: binary"
            }
            task.resume()
            _ = sem.wait(timeout: .now() + 12)
            return out
        }
        ctx.setObject(iosGet, forKeyedSubscript: "iosGet" as NSString)
        ctx.setObject(iosHas, forKeyedSubscript: "iosHas" as NSString)
        ctx.setObject(iosSet, forKeyedSubscript: "iosSet" as NSString)
        ctx.setObject(iosRemove, forKeyedSubscript: "iosRemove" as NSString)
        ctx.setObject(iosSleep, forKeyedSubscript: "iosSleep" as NSString)
        ctx.setObject(iosFetch, forKeyedSubscript: "iosFetch" as NSString)
        ctx.evaluateScript("""
        function __fmt(v) {
          if (v === undefined) return "undefined";
          if (v === null) return "null";
          if (typeof v === "object") { try { return JSON.stringify(v); } catch(e) { return String(v); } }
          return String(v);
        }
        function log() { iosLog(Array.from(arguments).map(__fmt).join(" ")); }
        const console = { log: log, info: log, warn: log, error: log, debug: log };
        function print() { log.apply(null, arguments); }
        function alert(msg) { log("ALERT: " + __fmt(msg)); }
        function prompt(msg, def) { log("PROMPT: " + __fmt(msg)); return def === undefined ? "" : String(def); }
        // Synchronous timer polyfill: runs the callback after a short blocking
        // sleep so template code using setTimeout still works in order.
        // (No background intervals on iOS JSContext — use Re-run for loops.)
        var __timerId = 0;
        function setTimeout(fn, ms) {
          __timerId += 1;
          try { iosSleep(Number(ms) || 0); } catch(e) {}
          try { if (typeof fn === "function") fn(); else eval(fn); } catch(e) { throw e; }
          return __timerId;
        }
        function clearTimeout(id) {}
        function setInterval(fn, ms) { return setTimeout(fn, ms); }
        function clearInterval(id) {}
        function btoa(s) { s = String(s); var chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/="; var out = "", i = 0; while (i < s.length) { var a = s.charCodeAt(i++), b = s.charCodeAt(i++), c = s.charCodeAt(i++); var x = (a << 16) | ((isNaN(b) ? 0 : b) << 8) | (isNaN(c) ? 0 : c); out += chars[(x >> 18) & 63] + chars[(x >> 12) & 63] + (isNaN(b) ? "=" : chars[(x >> 6) & 63]) + (isNaN(c) ? "=" : chars[x & 63]); } return out; }
        function atob(s) { s = String(s); var chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/="; var out = "", i = 0; s = s.replace(new RegExp("[^A-Za-z0-9+/=]", "g"), ""); while (i < s.length) { var e1 = chars.indexOf(s.charAt(i++)), e2 = chars.indexOf(s.charAt(i++)), e3 = chars.indexOf(s.charAt(i++)), e4 = chars.indexOf(s.charAt(i++)); var x = (e1 << 18) | (e2 << 12) | ((e3 & 63) << 6) | (e4 & 63); out += String.fromCharCode((x >> 16) & 255) + (e3 === 64 ? "" : String.fromCharCode((x >> 8) & 255)) + (e4 === 64 ? "" : String.fromCharCode(x & 255)); } return out; }
        function fetch(url) { var text = iosFetch(String(url)); return { text: text, json: function() { return JSON.parse(text); } }; }
        const localStorage = {
          getItem: function(k) { k = String(k); return iosHas(k) ? iosGet(k) : null; },
          setItem: function(k, v) { iosSet(String(k), String(v)); },
          removeItem: function(k) { iosRemove(String(k)); }
        };
        """)

        ctx.exceptionHandler = { _, exc in
            if let e = exc {
                handlerError = formatJSException(e)
            }
        }

        let value = ctx.evaluateScript(code)

        if let exc = ctx.exception {
            handlerError = formatJSException(exc)
            ctx.exception = nil
        }

        res.error = handlerError
        res.logs = Array(box.lines.prefix(200))
        if let v = value, !v.isUndefined {
            let s = v.toString() ?? ""
            // Hide bare "undefined" when there are already logs.
            if s != "undefined" || box.lines.isEmpty {
                res.result = s
            }
        }
        return res
    }
}

        try? mod.code.write(to: url, atomically: true, encoding: .utf8)
        return url
    }

    private func exportAll() {
        let arr = store.modules.map { ["name": $0.name, "code": $0.code, "pinned": $0.isPinned, "folder": $0.folder, "createdAt": $0.createdAt.timeIntervalSince1970] as [String: Any] }

            let url = FileManager.default.temporaryDirectory.appendingPathComponent("xcontainer-modules.json")

        }
    }

        lastRunText = ""
        quickRunner.start(mod.code)
    }

// Wrapper so Export-All can use .fileExporter.
struct ExportDoc: FileDocument {
    var url: URL?
    static var readableContentTypes: [UTType] { [.json] }
    init(url: URL?) { self.url = url }
    init(configuration: ReadConfiguration) throws { self.url = nil }
    func fileWrapper(configuration: WriteConfiguration) throws -> FileWrapper {
        if let url, let data = try? Data(contentsOf: url) {
            return FileWrapper(regularFileWithContents: data)
        }
        return FileWrapper(regularFileWithContents: Data("{}".utf8))
    }
}

// MARK: - Safe runner (JS runs off main thread, Stop + timeout included)
final class SafeRunner: ObservableObject {
    @Published var result: JSRunResult? = nil
    @Published var isRunning = false
    private var generation = 0

    func start(_ code: String, timeout: Double = 8) {
        generation += 1
        let g = generation
        isRunning = true
        result = nil
        DispatchQueue.global(qos: .userInitiated).async {
            let r = JSEngine.run(code)
            DispatchQueue.main.async {
                guard g == self.generation else { return }
                self.result = r
                self.isRunning = false
            }
        }
        DispatchQueue.main.asyncAfter(deadline: .now() + timeout) {
            guard g == self.generation && self.isRunning else { return }
            self.generation += 1
            self.isRunning = false
            var t = JSRunResult()
            t.error = "Timed out - likely an infinite loop. Tap Re-run to try again."
            self.result = t
        }
    }

    func stop() {
        generation += 1
        isRunning = false
    }
}

func looksLikeHTML(_ s: String) -> Bool {
    let lower = s.lowercased()
    guard s.contains("<") && s.contains(">") else { return false }
    let tags = ["<html", "<div", "<button", "<h1", "<h2", "<h3", "<p", "<style", "<span", "<ul", "<ol", "<li", "<input", "<table", "<img", "<a ", "<form"]
    return tags.contains(where: { lower.contains($0) })
}

struct HTMLView: UIViewRepresentable {
    let html: String
    func makeCoordinator() -> Coordinator { Coordinator() }
    func makeUIView(context: Context) -> WKWebView {
        let config = WKWebViewConfiguration()
        config.allowsInlineMediaPlayback = true
        config.preferences.javaScriptCanOpenWindowsAutomatically = true
        let v = WKWebView(frame: .zero, configuration: config)
        v.isOpaque = false
        v.backgroundColor = .clear
        v.scrollView.backgroundColor = .clear
        v.scrollView.bounces = true
        v.scrollView.isScrollEnabled = true
        return v
    }
    func updateUIView(_ view: WKWebView, context: Context) {
        // Don't reload on every SwiftUI refresh — that would wipe
        // scroll position / input text in the interactive panel.
        if context.coordinator.lastHTML != html {
            context.coordinator.lastHTML = html
            view.loadHTMLString(html, baseURL: nil)
        }
    }
    final class Coordinator {
        var lastHTML: String = ""
    }
}

            }

// MARK: - Module detail: tap a module -> see live output
// Split into small views so plain `swiftc` builds (BuildBox) type-check fast.
func runSummaryText(_ r: JSRunResult) -> String {
    let logsText: String = r.logs.joined(separator: " ")
    let parts: [String] = [logsText, r.result].filter { !$0.isEmpty }
    let joined: String = parts.joined(separator: " | ")
    if joined.isEmpty { return "(no output)" }
    return String(joined.prefix(120))
}

func runHadError(_ r: JSRunResult) -> Bool {
    guard let e = r.error else { return false }
    return !e.isEmpty
}

struct ModuleDetailView: View {
    @EnvironmentObject var store: ModuleStore
    let moduleID: UUID
    @StateObject private var runner = SafeRunner()

    private var module: Module? { store.modules.first(where: { $0.id == moduleID }) }

    var body: some View {
        ScrollView {
            detailStack
                .padding()
        }
        .navigationTitle("Module")
        .navigationBarTitleDisplayMode(.inline)
        .background(Color.black.ignoresSafeArea())
        .onAppear { autoRun() }
        .onChange(of: module?.code ?? "") { _, _ in autoRun() }
        .onChange(of: runner.result) { _, _ in recordCurrentRun() }
    }

    private func autoRun() {
        guard let m = module else { return }
        let trimmed = m.code.trimmingCharacters(in: .whitespacesAndNewlines)
        if !trimmed.isEmpty { runner.start(m.code) }
    }

    private func recordCurrentRun() {
        guard let r = runner.result else { return }
        guard let m = module else { return }
        store.recordRun(moduleName: m.name, summary: runSummaryText(r), hadError: runHadError(r))
    }

    @ViewBuilder
    private var detailStack: some View {
        VStack(alignment: .leading, spacing: 12) {
            if let m = module {
                detailHeader(m)
                detailButtons(m)
                detailStatus
            } else {
                Text("This module was deleted.").foregroundStyle(.secondary)
            }
        }
    }

    @ViewBuilder
    private func detailHeader(_ m: Module) -> some View {
        Text(m.name).font(.title2.bold())
        Text(m.code)
            .font(.system(.caption, design: .monospaced))
            .foregroundStyle(.secondary)
            .frame(maxWidth: .infinity, alignment: .leading)
            .padding(10)
            .background(Color(white: 0.08))
            .clipShape(RoundedRectangle(cornerRadius: 10))
    }

    @ViewBuilder
    private func detailButtons(_ m: Module) -> some View {
        HStack {
            Button("Re-run") { runner.start(m.code) }
                .buttonStyle(.borderedProminent)
                .disabled(runner.isRunning)
            if runner.isRunning {
                Button("Stop") { runner.stop() }
                    .buttonStyle(.bordered)
            }
            Button(m.isPinned ? "Unpin from Tabs" : "Pin to Tabs") {
                store.togglePin(m)
            }
            .buttonStyle(.bordered)
        }
    }

    @ViewBuilder
    private var detailStatus: some View {
        if runner.isRunning {
            Text("Running… (Stop frees the app if it loops)").foregroundStyle(.secondary)
        } else if let r = runner.result {
            RunOutputView(result: r)
        }
    }
}

        }
        .navigationTitle(module?.name ?? "Panel")
        .background(Color.black.ignoresSafeArea())
        .onAppear { autoRun() }
        .onChange(of: module?.code ?? "") { _, _ in autoRun() }
        .onChange(of: runner.result) { _, _ in recordCurrentRun() }
    }

    private func autoRun() {
        guard let m = module else { return }
        let trimmed = m.code.trimmingCharacters(in: .whitespacesAndNewlines)
        if !trimmed.isEmpty { runner.start(m.code) }
    }

    private func recordCurrentRun() {
        guard let r = runner.result else { return }
        guard let m = module else { return }
        store.recordRun(moduleName: m.name, summary: runSummaryText(r), hadError: runHadError(r))
    }

    @ViewBuilder
    private var panelStack: some View {
        VStack(alignment: .leading, spacing: 12) {
            if let m = module {
                Text(m.name).font(.title2.bold())
                panelButtons(m)
                panelStatus
            } else {
                Text("Unpinned or deleted.").foregroundStyle(.secondary)
            }
        }
    }

    @ViewBuilder
    private func panelButtons(_ m: Module) -> some View {
        HStack {
            Button("Re-run") { runner.start(m.code) }
                .buttonStyle(.borderedProminent)
                .disabled(runner.isRunning)
            if runner.isRunning {
                Button("Stop") { runner.stop() }
                    .buttonStyle(.bordered)
            }
        }
    }

    @ViewBuilder
    private var panelStatus: some View {
        if runner.isRunning {
            Text("Running…").foregroundStyle(.secondary)
        } else if let r = runner.result {
            RunOutputView(result: r)
        }
    }
}

// MARK: - Editor: type code -> Run -> Save (+ optional Pin as tab)
struct EditorView: View {
    @EnvironmentObject var store: ModuleStore
    @Binding var selection: String

    @State private var name = ""
    @State private var code = "// write your script here\nconsole.log(\"hello from XContainer\");\n1 + 2;"
    @State private var folder = "General"
    @State private var pinAsPanel = true
    @StateObject private var runner = SafeRunner()
    @State private var loadedID: UUID? = nil
    @State private var fontSize: Double = 15

    private var isEditing: Bool { loadedID != nil }

    var body: some View {
        ScrollView {
            VStack(alignment: .leading, spacing: 12) {
                Text(isEditing ? "Edit Module" : "Script Editor")
                    .font(.title2.bold())

                Text("Module name").font(.caption).foregroundStyle(.secondary)
                TextField("My first module", text: $name)
                    .textFieldStyle(.roundedBorder)

                Text("Container").font(.caption).foregroundStyle(.secondary)
                TextField("General", text: $folder)
                    .textFieldStyle(.roundedBorder)
                    .textInputAutocapitalization(.never)
                    .disableAutocorrection(true)
                ScrollView(.horizontal, showsIndicators: false) {
                    HStack {
                        ForEach(store.folders(), id: \.self) { f in
                            Button(f) { folder = f }
                                .font(.caption)
                                .buttonStyle(.bordered)
                        }
                    }
                }

                HStack {
                    Text("Script (JavaScript)").font(.caption).foregroundStyle(.secondary)
                    Spacer()
                    Text("\(code.components(separatedBy: "\n").count) lines").font(.caption).foregroundStyle(.secondary)
                    Button("A-") { fontSize = max(11, fontSize - 1) }.font(.caption)
                    Button("A+") { fontSize = min(22, fontSize + 1) }.font(.caption)
                    Button("Clear") { code = "" }.font(.caption)
                }
                TextEditor(text: $code)
                    .font(.system(size: fontSize, design: .monospaced))
                    .scrollContentBackground(.hidden)
                    .frame(minHeight: 220)
                    .padding(8)
                    .background(Color(white: 0.08))
                    .clipShape(RoundedRectangle(cornerRadius: 10))
                    .textInputAutocapitalization(.never)
                    .disableAutocorrection(true)

                ScrollView(.horizontal, showsIndicators: false) {
                    HStack {
                        snippetButton("log", insert: "console.log(\"hello\");\n")
                        snippetButton("loop", insert: "for (let i = 0; i < 5; i++) { console.log(i); }\n")
                        snippetButton("func", insert: "function name() {\n  return 1;\n}\n")
                        snippetButton("html", insert: "\"<div style='font-family:-apple-system;padding:20px;text-align:center'><h1>Hi</h1></div>\";\n")
                        snippetButton("store", insert: "localStorage.setItem(\"key\", \"value\");\n")
                        snippetButton("fetch", insert: "let r = fetch(\"https://api.open-meteo.com/v1/forecast?latitude=40.71&longitude=-74&current=temperature_2m\");\nconsole.log(r.text.slice(0,120));\n")
                    }
                }

                Toggle("Pin as tab panel on Save", isOn: $pinAsPanel)
                    .font(.caption)

                if let warn = store.validate(code: code) {
                    Text(warn).font(.caption).foregroundStyle(.orange)
                } else {
                    Text("Looks valid — ready to run.").font(.caption).foregroundStyle(.green)
                }
                HStack {
                    Button(runner.isRunning ? "Running…" : "Run") {
                        if store.validate(code: code) == nil { runner.start(code) }
                    }
                    .buttonStyle(.borderedProminent)
                    .frame(maxWidth: .infinity)
                    .disabled(runner.isRunning || store.validate(code: code) != nil)
                    if runner.isRunning {
                        Button("Stop") { runner.stop() }
                            .buttonStyle(.bordered)
                    }
                    Button("Save") {
                        save()
                    }
                    .buttonStyle(.bordered)
                    .frame(maxWidth: .infinity)
                    .disabled(name.trimmingCharacters(in: .whitespaces).isEmpty)
                }

                if isEditing {
                    Button("New blank script") {
                        store.editingModuleID = nil
                        loadedID = nil
                        name = ""
                        folder = "General"
                        code = "// write your script here"
                        runner.stop()
                    }
                    .font(.caption)
                }

                if let r = runner.result {
                    RunOutputView(result: r)
                }
            }
            .padding()
        }
        .navigationTitle("Editor")
        .background(Color.black.ignoresSafeArea())
        .onAppear(perform: loadIfNeeded)
        .onChange(of: store.editingModuleID) { _, _ in loadIfNeeded() }
    }

    private func loadIfNeeded() {
        let target = store.editingModuleID
        guard target != loadedID else { return }
        loadedID = target
        if let m = store.module(withID: target) {
            name = m.name
            code = m.code
            folder = m.folder
            pinAsPanel = m.isPinned
        } else {
            // Fresh script (Home -> Create, Modules -> +, or after Save).
            // Always reset so we never show the previous module's text.
            name = ""
            code = "// write your script here\nconsole.log(\"hello from XContainer\");\n1 + 2;"
            folder = "General"
            pinAsPanel = true
        }
        runner.stop()
    }

    private func snippetButton(_ label: String, insert: String) -> some View {
        Button(label) { code += insert }
            .font(.caption)
            .buttonStyle(.bordered)
    }

    private func save() {
        let clean = name.trimmingCharacters(in: .whitespaces)
        guard !clean.isEmpty else { return }
        // Empty code is allowed but confirm-ish: save a placeholder comment
        // so detail/panel never shows a blank confusing screen.
        let finalCode = code.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty ? "// empty module\nconsole.log(\"empty module\");" : code
        if let id = loadedID, store.module(withID: id) != nil {
            store.update(id: id, name: clean, code: finalCode, folder: folder)
            if let m = store.module(withID: id) { store.setPinned(m, pinned: pinAsPanel) }
        } else {
            let m = store.add(name: clean, code: finalCode, pinned: pinAsPanel, folder: folder)
            loadedID = m.id
            store.editingModuleID = m.id
        }
        store.editingModuleID = nil
        loadedID = nil
        name = ""
        folder = "General"
        code = "// write your script here\nconsole.log(\"hello from XContainer\");\n1 + 2;"
        runner.stop()
        selection = "modules"
    }
}

struct SettingsView: View {
    @EnvironmentObject var store: ModuleStore
    @State private var selfTest = ""
    @State private var showResetConfirm = false

    var body: some View {
        List {
            Section {
                Text("XContainer is a Swift scripting container. Write JavaScript modules in the Editor, run them on JavaScriptCore, save them, and pin them as their own tabs/panels.")
                    .font(.caption)
                    .foregroundStyle(.secondary)
            } header: { Text("About XContainer") }

            Section {
                Button("Add back example modules") { store.restoreExamples() }
                Text("Adds any missing examples. Your modules are never touched.")
                    .font(.caption2).foregroundStyle(.secondary)
                Button("Reset all modules", role: .destructive) { showResetConfirm = true }
            } header: { Text("Modules") }

            Section {
                if store.history.isEmpty {
                    Text("No runs yet. Run anything and it shows up here.")
                        .font(.caption).foregroundStyle(.secondary)
                } else {
                    ForEach(store.history.prefix(20)) { h in
                        VStack(alignment: .leading, spacing: 2) {
                            HStack {
                                Text(h.moduleName).font(.caption.bold())
                                Spacer()
                                Text(h.at, style: .time).font(.caption2).foregroundStyle(.secondary)
                            }
                            Text(h.summary).font(.caption2).foregroundStyle(.secondary).lineLimit(2)
                            if h.hadError { Text("error").font(.caption2).foregroundStyle(.red) }
                        }
                        .padding(.vertical, 2)
                    }
                    Button("Clear history", role: .destructive) { store.clearHistory() }
                }
            } header: { Text("Run history") }

            Section {
                Text("Modules are stored on-device (UserDefaults). Code runs locally with Apple's JavaScriptCore — no server.")
                    .font(.caption)
                    .foregroundStyle(.secondary)
                Button("Run engine self-test") { selfTest = JSEngine.selfTest() }
                if !selfTest.isEmpty {
                    Text(selfTest).font(.caption2).foregroundStyle(.secondary)
                }
            } header: { Text("Storage & Engine") }
        }
        .scrollContentBackground(.hidden)
        .navigationTitle("Settings")
        .alert("Delete ALL modules?", isPresented: $showResetConfirm) {
            Button("Delete everything", role: .destructive) { store.reset() }
            Button("Cancel", role: .cancel) {}
        } message: {
            Text("This wipes every module on this device, including history-backed names. Export first if unsure.")
        }
    }
}
